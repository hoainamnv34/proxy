#!/bin/bash

# MongoDB connection details
DB_NAME="cd"           # Tên database
COLLECTION_NAME="ab"   # Tên collection
MONGO_HOST="localhost" # Địa chỉ MongoDB
MONGO_PORT="27017"     # Cổng MongoDB

# Calculate the epoch time for one year ago
ONE_YEAR_AGO=$(($(date +%s) - 365*24*60*60))

# Construct the MongoDB query
QUERY="{\"event_time\": {\"\$lt\": $ONE_YEAR_AGO}}"

# Execute the deleteMany operation
mongo "$DB_NAME" --host "$MONGO_HOST" --port "$MONGO_PORT" --quiet --eval "
db.getCollection('$COLLECTION_NAME').deleteMany($QUERY);
print('Deleted documents older than one year.');
"

# Check exit status
if [ $? -eq 0 ]; then
  echo "Cleanup completed successfully."
else
  echo "Error during cleanup process."
fi
